# Additional permanent demo_dbo fixtures used after the equal-skill, no-armour baseline.

# A warding profile that can actually wield a two-handed reach weapon.
combat clone 15 "Polearm Warder (Auto)"
combat config setup twohand
combat config prefer_shield false
combat config global

# Repair the original pike fixture to use the compatible warding profile.
npc edit 110
npc set combatsetting "Polearm Warder (Auto)"
npc edit submit
npc review 110
accept edit

# Standard pike control, differing from #110 only by tactics.
npc clone 110
npc set unique "combat-human-mid-pike-standard"
npc set comment "Permanent human combat-balance fixture: mid-tier, infantry pike, no armour, Zweihander Auto."
npc set combatsetting "Zweihander (Auto)"
